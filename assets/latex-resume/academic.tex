% ASu-skills LaTeX 简历（学术风）
%
% 供 Overleaf 等环境继续编辑排版。
%
% ── Overleaf 使用步骤（无需本地安装任何 TeX 环境）─────────────────
%   1. New Project → Blank Project，填写 Project name → Create；
%   2. 上传本 .tex 文件；若简历使用证件照或校标，把图片一并上传到同一项目
%      （与 .tex 同级，文件名保持一致）；
%   3. 左上角 File → Settings：Main document 选本文件，
%      Compiler 选 XeLaTeX（中文必需）→ Recompile。

\documentclass[a4paper,11pt]{ctexart}

\usepackage[top=2cm,bottom=2cm,left=2.1cm,right=2.1cm]{geometry}
\usepackage{enumitem}
\usepackage{xcolor}
\usepackage{titlesec}
\usepackage[hidelinks]{hyperref}
\usepackage{graphicx}

% 学术版式不使用强调色，保证黑白打印一致
\definecolor{asurule}{gray}{0.35}

% 不使用 \scshape：小型大写是拉丁字母的字形变体，中文字体没有 sc 变体，
% 会触发 Font shape undefined 警告且对中文标题无效。学术观感改由无彩色、
% 字号、行距与页边距体现。
\titleformat{\section}
  {\normalfont\large\bfseries}{}{0pt}{}
  [{\color{asurule}\titlerule[0.5pt]}]
\titlespacing*{\section}{0pt}{14pt}{6pt}

\setlist[itemize]{leftmargin=1.4em,itemsep=2pt,parsep=0pt,topsep=4pt,partopsep=0pt}

\linespread{1.08}
\setlength{\parindent}{0pt}
\pagestyle{empty}

\newlength{\asuphotowidth}
\newlength{\asuheadtextwidth}

\newcommand{\asuphoto}[1]{%
  \IfFileExists{#1}%
    {\includegraphics[width=\dimexpr\asuphotowidth-4mm\relax]{#1}}%
    {\fbox{\parbox[c][30mm][c]{\dimexpr\asuphotowidth-4mm-2\fboxsep-2\fboxrule\relax}%
       {\centering\footnotesize 未找到图片\\\texttt{#1}\\请一并上传}}}%
}

\newcommand{\asuname}[1]{{\huge #1}\par\vspace{5pt}}
\newcommand{\asumeta}[1]{{\small #1}\par\vspace{1pt}}
\newcommand{\asuentry}[2]{\textbf{#1}\hfill{\small\itshape #2}\par}
