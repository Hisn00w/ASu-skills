\documentclass[a4paper,10pt]{ctexart}

\usepackage[top=1.7cm,bottom=1.7cm,left=1.9cm,right=1.9cm]{geometry}
\usepackage{enumitem}
\usepackage{xcolor}
\usepackage{titlesec}
\usepackage[hidelinks]{hyperref}
\usepackage{graphicx}

% 学术版式不使用强调色，保证黑白打印一致
\definecolor{asurule}{gray}{0.35}

% 不使用 \scshape：小型大写是拉丁字母的字形变体，中文字体没有 sc 变体，
% 会触发 Font shape undefined 警告且对中文标题无效。学术观感改由无彩色、
% 字号、行距与页边距体现。
\titleformat{\section}
  {\normalfont\large\bfseries}{}{0pt}{}
  [{\color{asurule}\titlerule[0.5pt]}]
\titlespacing*{\section}{0pt}{10pt}{4pt}

\setlist[itemize]{leftmargin=1.4em,itemsep=1pt,parsep=0pt,topsep=2pt,partopsep=0pt}

\linespread{1.04}
\setlength{\parindent}{0pt}
\pagestyle{empty}
